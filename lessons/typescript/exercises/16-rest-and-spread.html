Rest and spread

Collect many arguments or expand and merge arrays and objects.

How to Use This Page

Read the explanation first. Try the sample yourself, then compare the type choices and comments. Change one value at a time and predict the result before running it.

Examples

Method NameExplanationConcrete Sample
Rest and spreadCollect many arguments or expand and merge arrays and objects.

/****************Rest Operator***********************  */
function addNumber(...numbers: number[]): number {
    return numbers.reduce((acc, curr) => acc + curr, 0);
}

console.log(addNumber(1, 2, 3, 4)); // 10
console.log(addNumber(5, 10)); // 15
console.log(addNumber()); // 0


/****************Spread Operator***********************  */
let num1 = [1, 2, 3];
let num2 = [4, 5, 6];
//combine two arrays using spread operator
let combined = [...num1, ...num2];
console.log(combined); // [1, 2, 3, 4, 5, 6]

let combinedWithExtra = [...num1, ...num2, 7, 8];
console.log(combinedWithExtra); // [1, 2, 3, 4, 5, 6, 7, 8]

//spread operator with object   
let obj1 = { a: 1, b: 2 };
let obj2 = { c: 3, d: 4 };
let combinedObj = { ...obj1, ...obj2 };
console.log(combinedObj); // { a: 1, b: 2, c: 3, d: 4 }

let combinedObjWithExtraAndUpdate = { ...obj1, ...obj2, e: 5, f: 6, a:0 };
console.log(combinedObjWithExtraAndUpdate); // { a: 0, b: 2, c: 3, d: 4, e: 5, f: 6 }
