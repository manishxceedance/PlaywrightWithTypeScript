Enums

Represent a closed set of named values with numeric and string enum members.

How to Use This Page

Read the explanation first. Try the sample yourself, then compare the type choices and comments. Change one value at a time and predict the result before running it.

Examples

Method NameExplanationConcrete Sample
EnumsRepresent a closed set of named values with numeric and string enum members.
enum direction{
    North= "NORTH",
    South= "SOUTH",
    East= "EAST",
    West= "WEST"
}

console.log(direction);
console.log(direction.North);
console.log(direction.South);
console.log(direction.East);
console.log(direction.West);

// An enum is a group of named constants. It makes related values easier to read.

// 1. Numeric enum: values start at 0 and increase automatically.
enum Direction {
    North,
    South,
    East,
    West,
}

console.log('numeric enum:', Direction.North); // 0
console.log('numeric reverse lookup:', Direction[0]); // North

// 2. Numeric enum with custom starting values.
enum HttpStatus {
    Ok = 200,
    Created = 201,
    BadRequest = 400,
    NotFound = 404,
}

console.log('HTTP status:', HttpStatus.NotFound); // 404

// 3. String enum: every member has an explicit string value.
enum Role {
    Admin = 'ADMIN',
    Editor = 'EDITOR',
    Viewer = 'VIEWER',
}

const currentRole: Role = Role.Editor;
console.log('string enum:', currentRole); // EDITOR

// 4. Heterogeneous enum: numeric and string members can be mixed.
// This is valid, but it is usually clearer to use one value type consistently.
enum ApiResult {
    Success = 200,
    Failure = 'FAILURE',
}

console.log('heterogeneous enum:', ApiResult.Success, ApiResult.Failure);

// 5. Computed enum members: a member can be calculated from another value.
enum FilePermission {
    Read = 1,
    Write = 2,
    ReadWrite = Read | Write,
}

console.log('computed enum:', FilePermission.ReadWrite); // 3

// 6. Enum in a function parameter: only valid enum members are accepted.
function canEdit(role: Role): boolean {
    return role === Role.Admin || role === Role.Editor;
}

console.log('can edit:', canEdit(Role.Editor)); // true
console.log('can edit as viewer:', canEdit(Role.Viewer)); // false

// 7. Enum in a switch: useful when behavior depends on a named state.
function getDirectionText(value: Direction): string {
    switch (value) {
        case Direction.North:
            return 'Moving north';
        case Direction.South:
            return 'Moving south';
        case Direction.East:
            return 'Moving east';
        case Direction.West:
            return 'Moving west';
    }
}

console.log('switch example:', getDirectionText(Direction.East));

// 8. const enum: values are inserted at compile time, so no enum object is emitted.
const enum LogLevel {
    Info = 'INFO',
    Warning = 'WARNING',
    Error = 'ERROR',
}

const selectedLevel: LogLevel = LogLevel.Warning;
console.log('const enum:', selectedLevel);

// 9. Enum as a group of options for an object.
enum Theme {
    Light = 'light',
    Dark = 'dark',
}

type UserSettings = {
    username: string;
    theme: Theme;
};

const settings: UserSettings = {
    username: 'Manish',
    theme: Theme.Dark,
};

console.log('settings:', settings);

// 10. For simple string choices, a union type is often smaller than an enum.
type Size = 'small' | 'medium' | 'large';
const shirtSize: Size = 'medium';
console.log('union alternative:', shirtSize);
