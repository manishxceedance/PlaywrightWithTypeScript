Access modifiers

Control class access with public, private, and protected members.

How to Use This Page

Read the explanation first. Try the sample yourself, then compare the type choices and comments. Change one value at a time and predict the result before running it.

Examples

Method NameExplanationConcrete Sample
Access modifiersControl class access with public, private, and protected members.
class Car {
    public brand: string;
    protected model: string;
    private year: number;
    constructor(brand: string, model: string, year: number) {
        this.brand = brand;
        this.model = model;
        this.year = year;
    }
    displayModel() {
        console.log(`Brand: ${this.brand} Model: ${this.model} Year: ${this.year}`);
    }
}

let myCar: Car = new Car("Toyota", "Corolla", 2021);
myCar.brand= "Honda"; // Accessible because it's public
// myCar.model = "Civic"; // Error: 'model' is protected
// myCar.year = 2022; // Error: 'year' is private

class Maruti extends Car {
    public marutiDisplayModel() {
        console.log(`Brand: ${this.brand}`);
        console.log(`Model: ${this.model}`);
        // console.log(`Year: ${this.year}`);
    }
    constructor(brand: string, model: string, year: number) {
        super(brand, model, year);
    }
    displayModel() {
        console.log(`Model: ${this.model}`);
    }
    public superDisplayModel() {
        super.displayModel();
    }

}

let myMaruti: Maruti = new Maruti("Maruti", "Swift", 2022);
console.log("Displaying Maruti car details:");
myMaruti.marutiDisplayModel();
myMaruti.displayModel();
myMaruti.superDisplayModel();
