Arrays

Use typed lists and common collection operations safely.

How to Use This Page

Read the explanation first. Try the sample yourself, then compare the type choices and comments. Change one value at a time and predict the result before running it.

Examples

Method NameExplanationConcrete Sample
ArraysUse typed lists and common collection operations safely.
let user: string[] = ["Manish","Kumar","Gupta"];
console.log('user joined:', user.join(' '));

// Array methods reference
const numbers: number[] = [10, 20, 90, 80, 50, 40, 60, 70, 30];
console.log('numbers:', numbers); // TypeScript infers the type as number[]
console.log('typeof numbers:', typeof numbers); // Arrays are objects in JavaScript

// Static methods
console.log('Array.from:', Array.from('123')); // Creates an array from an iterable or array-like value
console.log('Array.isArray:', Array.isArray(numbers)); // Checks whether a value is an array
console.log('Array.of:', Array.of(10, 20, 30)); // Creates an array from its arguments

// Methods that read or search without changing the original array
console.log('at:', numbers.at(2)); // Returns the item at an index; negative indexes count from the end
console.log('concat:', numbers.concat([100, 110])); // Combines arrays or values into a new array
console.log('entries:', [...numbers.entries()]); // Returns an iterator containing [index, value] pairs
console.log('every:', numbers.every((number) => number > 0)); // Checks whether every item passes a test
console.log('find:', numbers.find((number) => number > 75)); // Returns the first item that passes a test
console.log('findIndex:', numbers.findIndex((number) => number > 75)); // Returns the index of the first matching item
console.log('findLast:', numbers.findLast((number) => number > 75)); // Returns the last item that passes a test
console.log('findLastIndex:', numbers.findLastIndex((number) => number > 75)); // Returns the index of the last matching item
console.log('includes:', numbers.includes(90)); // Checks whether an item exists in the array
console.log('indexOf:', numbers.indexOf(90)); // Returns the first index of an item, or -1
console.log('join:', numbers.join(' - ')); // Joins all items into a string using a separator
console.log('keys:', [...numbers.keys()]); // Returns an iterator containing the array indexes
console.log('lastIndexOf:', [10, 20, 10].lastIndexOf(10)); // Returns the last index of an item, or -1
console.log('slice:', numbers.slice(1, 4)); // Copies part of an array into a new array
console.log('some:', numbers.some((number) => number > 85)); // Checks whether at least one item passes a test
console.log('toString:', numbers.toString()); // Converts the array to a comma-separated string
console.log('values:', [...numbers.values()]); // Returns an iterator containing the array values

// Methods that transform an array and return a new array
console.log('filter:', numbers.filter((number) => number >= 50)); // Returns items that pass a test
console.log('flat:', [1, [2, [3]]].flat(2)); // Flattens nested arrays by the specified depth
console.log('map:', numbers.map((number) => number * 2)); // Creates a new array from transformed items
console.log('toReversed:', numbers.toReversed()); // Returns a reversed copy without changing the original
console.log('toSorted:', numbers.toSorted((first, second) => first - second)); // Returns a sorted copy without changing the original
console.log('with:', numbers.with(0, 999)); // Returns a copy with one item replaced

// Methods that execute a function or combine all items
numbers.forEach((number, index) => console.log(`forEach [${index}]:`, number)); // Runs a function once for each item
console.log('reduce:', numbers.reduce((total, number) => total + number, 0)); // Reduces all items to one value from left to right
console.log('reduceRight:', ['A', 'B', 'C'].reduceRight((text, letter) => text + letter, '')); // Reduces all items from right to left

// Methods that change the original array
const mutableNumbers = [3, 1, 2, 4];
console.log('copyWithin:', mutableNumbers.copyWithin(0, 1)); // Copies part of an array to another position
console.log('fill:', [1, 2, 3].fill(0, 1, 3)); // Replaces items with a value between start and end indexes
console.log('pop:', mutableNumbers.pop()); // Removes and returns the last item
mutableNumbers.push(4); // Adds one or more items to the end and returns the new length
console.log('after push:', mutableNumbers);
console.log('reverse:', mutableNumbers.reverse()); // Reverses the array in place
console.log('mutableNumbers after reverse:', mutableNumbers); // Check the array after reversing
console.log('shift:', mutableNumbers.shift()); // Removes and returns the first item
mutableNumbers.unshift(9); // Adds one or more items to the beginning and returns the new length
console.log('after unshift:', mutableNumbers);

let mutliTypeArray: (number | string | boolean)[] = [1, "two", true];
console.log('mutliTypeArray:', mutliTypeArray); // Logs the array with multiple types
console.log('mutliTypeArray values:', [...mutliTypeArray.values()]); // Logs the values of the array using the values() iterator
console.log('type of mutliTypeArray:', typeof mutliTypeArray); // Logs the type of the array
